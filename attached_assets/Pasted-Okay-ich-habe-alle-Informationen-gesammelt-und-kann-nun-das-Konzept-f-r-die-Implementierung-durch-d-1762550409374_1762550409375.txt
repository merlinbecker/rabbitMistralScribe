### Funktion der App
Die App ist eine Progressive Web App (PWA), die Audionotizen aufnimmt, transkribiert und zusammenfasst. Hier sind die Hauptfunktionen:

1. **Audioaufnahme**:
   - Aufnahme starten/stoppen mit dem "sideClick"-Event.
   - Frequenzanalyse und Visualisierung auf einem 16x16 Pixel-Display.

2. **Transkription**:
   - Lokale Speicherung der Aufnahme bis zur erfolgreichen Transkription.
   - Upload und Transkription bei verfügbarer Internetverbindung.
   - Verwendung der Mistral Voxtral API für die Transkription.

3. **Zusammenfassung**:
   - Weiterleitung des Transkripts an einen Mistral Agent für die Zusammenfassung nach einem bestimmten Schema.

4. **Speicherung**:
   - Speicherung des Rohtranskripts und der Zusammenfassung in einem GitHub Repository.

5. **Authentifizierung**:
   - Implementierung von OAuth2 und GitHub OAuth für die Authentifizierung.

6. **Optimierung**:
   - Optimierung für mobile Geräte, insbesondere für den Rabbit R1.

  ```javascript
  import { useState, useEffect } from 'react';

  const AudioRecorder = () => {
    const [isRecording, setIsRecording] = useState(false);
    const [audioChunks, setAudioChunks] = useState([]);
    const [mediaRecorder, setMediaRecorder] = useState(null);

    useEffect(() => {
      // Initialize media recorder
      const initMediaRecorder = async () => {
        try {
          const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
          const recorder = new MediaRecorder(stream);
          setMediaRecorder(recorder);

          recorder.ondataavailable = (event) => {
            setAudioChunks(prev => [...prev, event.data]);
          };

          recorder.onstop = () => {
            const audioBlob = new Blob(audioChunks, { type: 'audio/mp3' });
            // Save to local storage or send to server
          };
        } catch (error) {
          console.error('Error initializing media recorder:', error);
        }
      };

      initMediaRecorder();

      return () => {
        if (mediaRecorder) {
          mediaRecorder.stop();
        }
      };
    }, []);

    const startRecording = () => {
      if (mediaRecorder) {
        mediaRecorder.start();
        setIsRecording(true);
      }
    };

    const stopRecording = () => {
      if (mediaRecorder) {
        mediaRecorder.stop();
        setIsRecording(false);
      }
    };

    // Handle sideClick event for Rabbit R1
    useEffect(() => {
      window.addEventListener("sideClick", () => {
        if (isRecording) {
          stopRecording();
        } else {
          startRecording();
        }
      });

      return () => {
        window.removeEventListener("sideClick", () => {});
      };
    }, [isRecording]);

    return (
      <div>
        <button onClick={isRecording ? stopRecording : startRecording}>
          {isRecording ? 'Stop Recording' : 'Start Recording'}
        </button>
      </div>
    );
  };

  export default AudioRecorder;
  ```

- **FrequencyVisualizer.js**:
  ```javascript
  import { useState, useEffect } from 'react';

  const FrequencyVisualizer = () => {
    const [frequencyData, setFrequencyData] = useState(new Uint8Array(16));

    useEffect(() => {
      const audioContext = new (window.AudioContext || window.webkitAudioContext)();
      const analyser = audioContext.createAnalyser();
      analyser.fftSize = 256;

      const bufferLength = analyser.frequencyBinCount;
      const dataArray = new Uint8Array(bufferLength);

      const updateFrequencyData = () => {
        analyser.getByteFrequencyData(dataArray);
        const averageData = new Uint8Array(16);
        for (let i = 0; i < 16; i++) {
          const start = Math.floor(i * bufferLength / 16);
          const end = Math.floor((i + 1) * bufferLength / 16);
          let sum = 0;
          for (let j = start; j < end; j++) {
            sum += dataArray[j];
          }
          averageData[i] = Math.floor(sum / (end - start));
        }
        setFrequencyData(averageData);
        requestAnimationFrame(updateFrequencyData);
      };

      updateFrequencyData();

      return () => {
        audioContext.close();
      };
    }, []);

    return (
      <div id="pixel-display">
        {Array.from({ length: 16 * 16 }).map((_, index) => {
          const row = Math.floor(index / 16);
          const col = index % 16;
          const frequency = frequencyData[col];
          const color = frequency > 128 ? 'red' : frequency > 64 ? 'yellow' : 'orange';
          return <div key={index} className="pixel" style={{ backgroundColor: color }}></div>;
        })}
      </div>
    );
  };

  export default FrequencyVisualizer;
  ```

  ```javascript
  import axios from 'axios';

  const API_BASE_URL = 'https://api.mistral.ai';

  export const transcribeAudio = async (audioBlob) => {
    const formData = new FormData();
    formData.append('audio', audioBlob);

    try {
      const response = await axios.post(`${API_BASE_URL}/transcribe`, formData, {
        headers: {
          'Content-Type': 'multipart/form-data',
          'Authorization': `Bearer ${process.env.MISTRAL_API_KEY}`
        }
      });
      return response.data;
    } catch (error) {
      console.error('Error transcribing audio:', error);
      throw error;
    }
  };

  export const summarizeText = async (text) => {
    try {
      const response = await axios.post(`${API_BASE_URL}/summarize`, { text }, {
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${process.env.MISTRAL_API_KEY}`
        }
      });
      return response.data;
    } catch (error) {
      console.error('Error summarizing text:', error);
      throw error;
    }
  };
  ```

- **storage.js**:
  ```javascript
  export const saveToLocalStorage = (key, data) => {
    localStorage.setItem(key, JSON.stringify(data));
  };

  export const getFromLocalStorage = (key) => {
    const data = localStorage.getItem(key);
    return data ? JSON.parse(data) : null;
  };

  export const saveToGitHub = async (githubToken, repo, path, data) => {
    try {
      const response = await fetch(`https://api.github.com/repos/${repo}/contents/${path}`, {
        method: 'PUT',
        headers: {
          'Authorization': `token ${githubToken}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          message: 'Update audio notes',
          content: btoa(JSON.stringify(data))
        })
      });
      return response.json();
    } catch (error) {
      console.error('Error saving to GitHub:', error);
      throw error;
    }
  };
  ```

```javascript
import axios from 'axios';

const API_BASE_URL = 'https://api.mistral.ai';

export const transcribeAudio = async (audioBlob) => {
  const formData = new FormData();
  formData.append('audio', audioBlob);

  try {
    const response = await axios.post(`${API_BASE_URL}/transcribe`, formData, {
      headers: {
        'Content-Type': 'multipart/form-data',
        'Authorization': `Bearer ${process.env.MISTRAL_API_KEY}`
      }
    });
    return response.data;
  } catch (error) {
    console.error('Error transcribing audio:', error);
    throw error;
  }
};

export const summarizeText = async (text) => {
  try {
    const response = await axios.post(`${API_BASE_URL}/summarize`, { text }, {
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${process.env.MISTRAL_API_KEY}`
      }
    });
    return response.data;
  } catch (error) {
    console.error('Error summarizing text:', error);
    throw error;
  }
};
```

```javascript
export const saveToGitHub = async (githubToken, repo, path, data) => {
  try {
    const response = await fetch(`https://api.github.com/repos/${repo}/contents/${path}`, {
      method: 'PUT',
      headers: {
        'Authorization': `token ${githubToken}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        message: 'Update audio notes',
        content: btoa(JSON.stringify(data))
      })
    });
    return response.json();

  }
```
